
\section{Funkcionalni zahtevi}

Funkcionalni zahtevi definišu funkcionalnosti koje
informacioni sistem mora da obezbedi svojim korisnicima.
Na osnovu analize poslovanja rent-a-car agencije definisani
su sledeći funkcionalni zahtevi i podeljeni u grupacije.

\subsection{Upravljanje korisničkim nalozima}

\begin{itemize}
    \item FR01 - Sistem omogućava registraciju novih kupaca.
    \item FR02 - Sistem omogućava prijavljivanje i odjavljivanje korisnika.
    \item FR03 - Sistem omogućava zaposlenima pregled i izmenu podataka o kupcima.
    \item FR04 - Sistem omogućava administratoru kreiranje i upravljanje nalozima zaposlenih.
\end{itemize}

\subsection{Upravljanje vozilima}

\begin{itemize}
    \item FZ05 - Sistem omogućava pregled svih vozila u ponudi agencije.
    \item FZ06 - Sistem omogućava pretragu i filtriranje vozila prema zadatim kriterijumima.
    \item FZ07 - Sistem omogućava zaposlenima dodavanje novih vozila.
    \item FZ08 - Sistem omogućava zaposlenima izmenu i uklanjanje vozila iz ponude.
    \item FZ09 - Sistem omogućava evidentiranje tehničke ispravnosti i stanja vozila.
\end{itemize}

\subsection{Upravljanje rezervacijama}

\begin{itemize}
    \item FZ10 - Sistem omogućava kupcima kreiranje rezervacija.
    \item FZ11 - Sistem proverava dostupnost vozila za izabrani vremenski period.
    \item FZ12 - Sistem omogućava kupcima pregled i otkazivanje sopstvenih rezervacija.
    \item FZ13 - Sistem omogućava zaposlenima pregled i upravljanje rezervacijama.
\end{itemize}

\subsection{Upravljanje iznajmljivanjem vozila}

\begin{itemize}
    \item FZ14 - Sistem omogućava zaposlenima evidentiranje preuzimanja vozila.
    \item FZ15 - Sistem omogućava zaposlenima evidentiranje vraćanja vozila.
    \item FZ16 - Sistem omogućava evidentiranje oštećenja i promena stanja vozila.
    \item FZ17 - Sistem omogućava evidentiranje kašnjenja u vraćanju vozila.
\end{itemize}
